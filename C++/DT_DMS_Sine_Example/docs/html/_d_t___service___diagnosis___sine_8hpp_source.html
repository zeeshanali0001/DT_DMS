63 Output executeService() override {
64 auto spline = std::dynamic_pointer_cast<Models::DT_Model_Descr_CubicBSpline>(model);
65 if (consumedData.empty() || !spline) {
66 diagnosisReport = "Diagnosis failed: no data or no Cubic B-Spline model.";
67 return diagnosisReport;
68 }
69 double d = consumedData.back();
70 externalTransition(Event("data", d), spline->getStep());
71 std::vector<Event> emitted = internalTransition();
72 if (emitted.empty()) {
73 diagnosisReport = "OK";
74 } else {
75 double t = std::round(spline->getLastTime() * 1000.0) / 1000.0;
76 anomalyTimes.push_back(t);
77 diagnosisReport = "ANOMALY DETECTED at t=" + Automaton::formatDouble(t) + ": actual=" + Automaton::formatDouble(d) +
78 ", expected=" + Automaton::formatDouble(spline->getSineModelValue(spline->getLastTime())) +
79 ", residual=" + Automaton::anyToString(emitted.front().value);
80 }
81 return diagnosisReport;
82 }

93 double performRandomCheck(int numberOfPointsToCheck, unsigned int seed, bool verbose) const {
94 auto spline = std::dynamic_pointer_cast<Models::DT_Model_Descr_CubicBSpline>(model);
95 if (!spline || std::isnan(spline->getLastTime())) return std::numeric_limits<double>::quiet_NaN();
96 std::mt19937 generator(seed);
97 std::uniform_real_distribution<double> abscissa(spline->getLeftEndpoint(), spline->getLastTime());
98 double maxDeviation = 0.0;
99 for (int i = 0; i < numberOfPointsToCheck; ++i) {
100 double x = abscissa(generator);
101 double expected = std::sin(x);
102 double interpolated = spline->getSineModelValue(x);
103 maxDeviation = std::max(maxDeviation, std::fabs(expected - interpolated));
104 if (verbose) {
105 char line[160];
106 std::snprintf(line, sizeof(line),
107 " sin(%.6f) = %+.9f spline interpolation gives %+.9f delta = %.3e", x, expected,
108 interpolated, expected - interpolated);
109 std::cout << line << "\n";
110 }
111 }
112 return maxDeviation;
113 }

147 void specifyAutomaton() {
148 auto residual = [](const Automaton::State& q) { return std::any_cast<double>(q.at("residual")); };
149 addParameter("threshold", threshold); // Λ
150 addInput("data"); // X
151 addOutput("anomaly"); // Y
152 addStateVariable("residual", 0.0); // Σ
153 addPhase(Phase("IDLE", [residual](const Automaton::State& q) { return residual(q) >= 0; }, nullptr, // Φ
154 infinity()));
155 addPhase(Phase("CHECK", [residual](const Automaton::State& q) { return residual(q) >= 0; }, nullptr, 0.0));
156 addTransition(Transition(TransitionKind::In, "IDLE", "CHECK", {"data"}, nullptr, nullptr, // Δ^in
157 [this](Automaton::State q, const Event* x) {
158 q["residual"] = residualOf(std::any_cast<double>(x->value));
159 return q;
160 }));
161 addTransition(Transition(TransitionKind::Time, "CHECK", "IDLE", {}, // Δ^time (anomaly)
162 [this, residual](const Automaton::State& q) { return residual(q) > threshold; },
163 [](const Automaton::State& q) { return std::vector<Event>{{"anomaly", q.at("residual")}}; },
164 nullptr));
165 addTransition(Transition(TransitionKind::Time, "CHECK", "IDLE", {}, // Δ^time (normal)
166 [this, residual](const Automaton::State& q) { return residual(q) <= threshold; },
167 nullptr, nullptr));
168 setInitialization(Initialization("IDLE", nullptr, 0.0)); // Δ*
169 initialize();
170 }
