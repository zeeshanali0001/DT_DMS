61 void writeSvg(const std::string& fileName) const {
62 std::ofstream out(fileName, std::ios::binary);

64 const double width = 900, height = 600, left = 70, right = 30, top = 50, bottom = 60;
65 double xMin = xData.empty() ? 0 : xData.front(), xMax = xData.empty() ? 1 : xData.back();
66 double yMin = -1.0, yMax = 1.0;
67 for (std::size_t i = 0; i < xData.size(); ++i) {
68 for (double v : {rawSineData[i], modelData[i]}) {

79 auto py = [&](double y) { return top + (yMax - y) / (yMax - yMin) * (height - top - bottom); };
