48 using Automaton::formatDouble;
49 if (std::isnan(currentValue)) return std::string("STATUS: no value received yet for " + monitoredEntity + ".");
50 if (currentValue > upperThreshold)

52 " exceeds the upper threshold " + formatDouble(upperThreshold) + "!");
53 if (currentValue < lowerThreshold)
54 return std::string("ALERT: " + monitoredEntity + " = " + formatDouble(currentValue) +
55 " falls below the lower threshold " + formatDouble(lowerThreshold) + "!");
56 return std::string("STATUS: " + monitoredEntity + " = " + formatDouble(currentValue) + " is within safe limits.");
57 }
