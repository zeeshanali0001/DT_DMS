127 Kind kind(double tNow, double delta) const {
128 if (isEmpty()) throw std::invalid_argument("The empty segment cannot be classified.");
129 if (supDom() > tNow)
130 throw std::invalid_argument("No datum may lie in the future: sup Dom(ω) = " +
131 Automaton::formatDouble(supDom()) + " > t_now = " + Automaton::formatDouble(tNow));
132 if (isRealTime(tNow, delta)) return Kind::REAL_TIME;
133 return isHistorical(tNow, delta) ? Kind::HISTORICAL : Kind::MIXED;
134 }

159 DT_DataSegment<T> concat(const DT_DataSegment<T>& next) const {
160 if (!isEmpty() && !next.isEmpty() && supDom() >= next.infDom())
161 throw std::invalid_argument("ω1 ⊕ ω2 requires Dom(ω1) to precede Dom(ω2).");
162 DT_DataSegment<T> result = *this;
163 result.values.insert(next.values.begin(), next.values.end());
164 return result;
