57template <typename V>
58double numericValue(const V& d) {
59 if constexpr (std::is_arithmetic_v<V>)
60 return static_cast<double>(d);
61 else
62 return std::numeric_limits<double>::quiet_NaN();
63}

66template <typename V>
67std::string valueToString(const V& d) {
68 if constexpr (std::is_same_v<V, std::any>) {
69 return anyToString(d);
70 } else if constexpr (std::is_floating_point_v<V>) {
71 return formatDouble(static_cast<double>(d));
72 } else {
73 std::ostringstream out;
74 out << d;
75 return out.str();
