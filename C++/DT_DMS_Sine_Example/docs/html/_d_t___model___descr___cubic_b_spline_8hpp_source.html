60 double getSineModelValue(double x) const {
61 if (controlPoints.empty()) return std::numeric_limits<double>::quiet_NaN();
62 double s = (x - leftEndpoint) / step;
63 int i = static_cast<int>(std::floor(s));
64 double u = s - i;
65 double u2 = u * u;
66 double u3 = u2 * u;
67 return ((1 - u) * (1 - u) * (1 - u) * point(i - 1) + (3 * u3 - 6 * u2 + 4) * point(i) +
68 (-3 * u3 + 3 * u2 + 3 * u + 1) * point(i + 1) + u3 * point(i + 2)) /
69 6.0;
70 }

116 }));
