37 typename DT_Service<T>::Output executeService() override {
38 using Automaton::formatDouble;
39 auto prescriptive = std::dynamic_pointer_cast<Models::DT_Model_Prescriptive<T>>(this->model);
40 if (!prescriptive) {
41 controlActionResult = "Control action failed: no prescriptive model is coupled to the service.";
42 } else if (this->consumedData.empty()) {
43 controlActionResult = "Control action failed due to lack of inputs.";
44 } else {
45 double measured = Automaton::numericValue(this->consumedData.back());
46 double command = gain * (prescriptive->getTarget() - measured);
47 controlActionResult = "Control action (" + controlStrategy + "): command = " + formatDouble(command) +
48 " for measured value " + formatDouble(measured) + " and target " +
49 formatDouble(prescriptive->getTarget()) + ".";
50 }
51 return controlActionResult;
52 }
