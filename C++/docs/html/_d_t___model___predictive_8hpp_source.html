46 virtual double predict() const {
47 if (history.empty()) return std::numeric_limits<double>::quiet_NaN();
48 std::size_t from = history.size() > static_cast<std::size_t>(window) ? history.size() - window : 0;
49 double sum = 0.0;
50 for (std::size_t i = from; i < history.size(); ++i) sum += history[i];
51 return sum / static_cast<double>(history.size() - from);
52 }
