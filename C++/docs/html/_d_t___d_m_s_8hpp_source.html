167 void addService(const std::shared_ptr<Services::DT_Service<T>>& s, const std::shared_ptr<Models::DT_Model<T>>& m) {
168 if (!std::dynamic_pointer_cast<Services::DT_Service_Monitoring<T>>(s) &&

176 if (m) {
177 addModel(m);
178 m->addServiceToBeUpdated(s);
179 }
180 s->setModel(m);
181 if (std::find(serviceList.begin(), serviceList.end(), s) == serviceList.end()) serviceList.push_back(s);

237 bool checkServicesVersusModels(std::ostream& out) const {
238 bool ok = true;
239 out << "Services versus models:\n";
240 for (const auto& s : serviceList) {
241 const auto& m = s->getModel();
242 std::string problem;
243 bool needsPrescriptive = std::dynamic_pointer_cast<Services::DT_Service_Optimization<T>>(s) ||
244 std::dynamic_pointer_cast<Services::DT_Service_Control<T>>(s);
245 if (m && std::find(modelList.begin(), modelList.end(), m) == modelList.end())
246 problem = "its model " + m->getModelName() + " is not in M";
247 else if (needsPrescriptive && !std::dynamic_pointer_cast<Models::DT_Model_Prescriptive<T>>(m))
248 problem = "it needs a prescriptive model";
249 else if (std::dynamic_pointer_cast<Services::DT_Service_Prediction<T>>(s) && m &&
250 !std::dynamic_pointer_cast<Models::DT_Model_Predictive<T>>(m))
251 problem = "it is coupled to a model that is not predictive";
252 std::string model = m ? m->getModelName() : "no model (data-driven)";
253 if (problem.empty()) {
254 out << " OK " << s->getServiceName() << " → " << model << "\n";
255 } else {
256 out << " ERROR " << s->getServiceName() << " → " << model << ": " << problem << "\n";
257 ok = false;
258 }
259 }
260 return ok;
261 }

276 bool checkModelsVersusData(std::ostream& out, double epsilon = 1e-3, double bigM = 0.1) const {
277 using Automaton::formatDouble;
278 bool ok = true;
279 out << "Models versus data (ε=" << formatDouble(epsilon) << ", M=" << formatDouble(bigM) << "):\n";
280 for (const auto& m : modelList) {
281 for (const auto& d : dataList) {
282 double max = 0.0;
283 double sum = 0.0;
284 int count = 0;
285 for (const auto& value : d->getRealTimeData(delta).getValues()) {
286 double measured = Automaton::numericValue(value.second);
287 double expected = m->evaluate(value.first);
288 if (!std::isnan(measured) && !std::isnan(expected)) {
289 double deviation = std::fabs(measured - expected);
290 max = std::max(max, deviation);
291 sum += deviation;
292 count++;
293 }
294 }
295 std::string pair = m->getModelName() + " / " + d->getSource();
296 if (count == 0) {
297 out << " SKIP " << pair << ": the model gives no value at the real-time data\n";
298 } else if (max < epsilon) {
299 out << " OK " << pair << ": fits very well (max δ=" << formatDouble(max) << ")\n";
300 } else if (sum / count > bigM) {
301 out << " ERROR " << pair << ": does not fit (mean δ=" << formatDouble(sum / count)
302 << "): update, re-parameterize or discard the model\n";
303 ok = false;
304 } else {
305 out << " OK " << pair << ": acceptable (mean δ=" << formatDouble(sum / count)
306 << ", max δ=" << formatDouble(max) << ")\n";
307 }
308 }
309 }
310 return ok;
311 }

320 bool checkDataVersusServices(std::ostream& out) const {
321 bool ok = true;
322 out << "Data versus services:\n";
323 if (!serviceList.empty() && dataList.empty()) {
324 out << " ERROR the DT has services but no data\n";
325 ok = false;
326 }
327 for (const auto& d : dataList) {
328 if (!d->validateData()) {
329 out << " ERROR " << d->toString() << ": source or data type missing\n";
330 ok = false;
331 } else if (d->getCharacterization().getUnit().empty()) {
332 out << " WARN " << d->getSource() << ": no unit of measurement\n";
333 } else {
334 out << " OK " << d->getSource() << " (" << d->getDataType() << ", " << d->getCharacterization().getUnit()
335 << ")\n";
336 }
337 }
338 for (const auto& s : getMonitoringServices()) {
339 bool found = false;
340 for (const auto& d : dataList)
341 found = found || s->getMonitoredEntity() == d->getSource() || s->getMonitoredEntity() == d->getDataType();
342 if (!found)
343 out << " WARN " << s->getServiceName() << ": no data source for the monitored entity "
344 << s->getMonitoredEntity() << "\n";
345 }
346 return ok;
347 }

353 bool checkConsistency(std::ostream& out) const {
354 bool servicesModels = checkServicesVersusModels(out);
355 bool modelsData = checkModelsVersusData(out);
356 bool dataServices = checkDataVersusServices(out);

487 template <typename U, typename V>
488 static std::vector<std::shared_ptr<U>> filter(const std::vector<std::shared_ptr<V>>& list) {
489 std::vector<std::shared_ptr<U>> result;
490 for (const auto& element : list)
491 if (auto typed = std::dynamic_pointer_cast<U>(element)) result.push_back(typed);
492 return result;

496 static std::string block(const std::string& id, const std::string& title, const std::vector<std::string>& lines,
497 const std::string& color) {
498 std::string label;
499 for (std::size_t i = 0; i < lines.size(); ++i) label += (i > 0 ? " \\n " : "") + escape(lines[i]);
500 if (lines.empty()) label = "(none)";

502 }

505 static std::string escape(const std::string& s) {
506 std::string result;
507 for (char c : s) {
508 if (c == '\\' || c == '"') result += '\\';
509 result += c;
510 }
511 return result;
512 }
