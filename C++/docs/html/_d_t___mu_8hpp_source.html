101 void assimilate(const Data::DT_DataSegment<T>& omega) {
102 double tNow = clock();
103 switch (omega.kind(tNow, delta)) {
104 case Data::DT_DataSegment<T>::Kind::REAL_TIME:
105 assimilateSynchronously(omega);
106 break;
107 case Data::DT_DataSegment<T>::Kind::HISTORICAL:
108 assimilateAsynchronously(omega);
109 break;
110 case Data::DT_DataSegment<T>::Kind::MIXED:
111 switch (mixedPolicy) {
112 case MixedPolicy::SPLIT:
113 // ω = ω_hist ⊕ ω_rt, split at t_now − δ
114 assimilateAsynchronously(omega.historicalPart(tNow, delta));
115 assimilateSynchronously(omega.realTimePart(tNow, delta));
116 break;
117 case MixedPolicy::AS_REAL_TIME:
118 assimilateSynchronously(omega);
119 break;
120 case MixedPolicy::AS_HISTORICAL:
121 assimilateAsynchronously(omega);
122 break;
123 }
124 break;
125 }
126 }
127
130 void assimilateSynchronously(const Data::DT_DataSegment<T>& omega) {
131 for (const auto& m : models) mu(omega, *m);
132 }
133
136 void assimilateAsynchronously(const Data::DT_DataSegment<T>& omega) { pending.push_back(omega); }
137
139 void flush() {
140 std::vector<Data::DT_DataSegment<T>> toAssimilate;
141 toAssimilate.swap(pending);
142 for (const auto& omega : toAssimilate) assimilateSynchronously(omega);
143 }
