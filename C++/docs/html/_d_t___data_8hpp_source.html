85 // Safe copy to avoid issues if an observer removes itself during update

101 void updateData(double t, const T& newData) {
102 double tNow = clock();
103 if (t > tNow)
104 throw std::invalid_argument("No datum may lie in the future: t = " + Automaton::formatDouble(t) +
105 " > t_now = " + Automaton::formatDouble(tNow));
106 currentData = newData;
107 currentTime = t;
108 collectedData.put(t, newData);
109 notifyObservers();
110 }
